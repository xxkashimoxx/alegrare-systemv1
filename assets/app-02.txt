function openOpportunities() { return state.opportunities.filter((o) => o.status !== 'Recuperada'); }
function totalPotential() { return openOpportunities().reduce((sum, o) => sum + o.value, 0); }
function highPriority() { return openOpportunities().filter((o) => o.priority === 'Alta'); }
function recoveryCandidates() { return state.patients.filter((p) => ['Sem retorno','Faltou e não reagendou','Orçamento pendente','Tratamento interrompido','Retorno vencido'].includes(p.status)); }

function dashboardPage() {
  const open = openOpportunities();
  const potential = totalPotential();
  const top = [...open].sort((a,b) => b.value - a.value).slice(0,3);
  const concentration = potential ? Math.round(top.reduce((s,o)=>s+o.value,0) / potential * 100) : 0;
  return `
    <section class="hero-head">
      <div><span class="eyebrow">PAINEL INTELIGENTE · ATUALIZADO AGORA</span><h1>Olá, Danielle.</h1><p>O painel prioriza o que merece ação hoje — não apenas o que aconteceu.</p></div>
      <div class="hero-actions"><button class="btn secondary" data-route="opportunities">Ver oportunidades</button><button class="btn primary" data-action="new-prescription">+ Nova prescrição</button></div>

    <section class="metric-grid">
      ${metric('calendar','Consultas hoje','9','2 aguardando confirmação','neutral')}
