function opportunityBars() {
  const groups = {};
  openOpportunities().forEach(o => groups[o.category] = (groups[o.category] || 0) + o.value);
  const max = Math.max(...Object.values(groups), 1);
  return `<div class="bar-list">${Object.entries(groups).sort((a,b)=>b[1]-a[1]).map(([label,value]) => `<div class="bar-item"><div><span>${label}</span><b>${money(value)}</b></div><div class="bar-track"><i style="width:${Math.max(8, value/max*100)}%"></i></div></div>`).join('')}</div>`;
}
